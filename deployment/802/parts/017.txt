export function kaplanMeier(time,event,alpha=.05){clean(time,2);assert(time.every(x=>x>=0)&&event.length===time.length&&event.every(x=>x===0||x===1),'Survival time must be ≥0 and event must be exactly 0 or 1.');const times=[...new Set(time)].sort((a,b)=>a-b),z=zCritical(alpha);let S=1,g=0;const rows=times.map(t=>{const risk=time.filter(x=>x>=t).length,indices=time.map((x,i)=>x===t?i:-1).filter(i=>i>=0),d=sum(indices.map(i=>event[i])),c=indices.length-d;S*=1-d/risk;if(d>0&&risk>d)g+=d/(risk*(risk-d));let ci;if(S===1)ci=[1,1];else if(S===0)ci=[0,0];else{const q=Math.log(-Math.log(S)),se=Math.sqrt(g)/Math.abs(Math.log(S));ci=[Math.exp(-Math.exp(q+z*se)),Math.exp(-Math.exp(q-z*se))];}return{time:t,risk,events:d,censored:c,survival:S,ci};});return{n:time.length,events:sum(event),censored:event.length-sum(event),median:rows.find(r=>r.survival<=.5)?.time??null,rows,note:'Kaplan–Meier with Greenwood log-log pointwise intervals; non-informative right censoring assumed. Events precede censoring within tied times.'};}
export function logrank(t1,e1,t2,e2){kaplanMeier(t1,e1);kaplanMeier(t2,e2);const times=[...new Set(t1.filter((_,i)=>e1[i]).concat(t2.filter((_,i)=>e2[i])))].sort((a,b)=>a-b);let O=0,E=0,V=0;for(const t of times){const n1=t1.filter(x=>x>=t).length,n2=t2.filter(x=>x>=t).length,n=n1+n2,d1=t1.filter((x,i)=>x===t&&e1[i]).length,d2=t2.filter((x,i)=>x===t&&e2[i]).length,d=d1+d2;O+=d1;E+=d*n1/n;if(n>1)V+=n1*n2*d*(n-d)/(n*n*(n-1));}assert(V>0,'Log-rank variance is zero: no informative event contrast.');const chi2=(O-E)**2/V;return{observed:O,expected:E,variance:V,chi2,df:1,p:chiP(chi2,1),note:'Two-group unweighted log-rank; no hazard ratio inferred.'};}
export function series(x,y,maxLag=20){clean(x,3);clean(y,3);assert(x.length===y.length,'Align time and response observations.');const dt=x.slice(1).map((v,i)=>v-x[i]);assert(dt.every(x=>x>0),'Time must be strictly increasing with no duplicate times.');const h=mean(dt),regular=dt.every(d=>Math.abs(d-h)<1e-6*Math.max(1,Math.abs(h))),m=mean(y),den=sum(y.map(v=>(v-m)**2)),acf=[];if(regular&&den>0)for(let k=0;k<=Math.min(maxLag,Math.floor(y.length/3));k++)acf.push({lag:k,timeLag:k*h,r:sum(y.slice(k).map((v,i)=>(v-m)*(y[i]-m)))/den});return{...describe(y),duration:x.at(-1)-x[0],area:sum(y.slice(1).map((v,i)=>(v+y[i])*dt[i]/2)),regular,step:h,acf,note:regular?'ACF is descriptive; stationarity is not established. A trajectory has dependent samples.':'Irregular sampling: ACF is withheld; no implicit interpolation was performed.'};}
export function powerCurve(d=.5,alpha=.05,target=.8){assert(d>0&&d<=10&&target>.5&&target<1,'Use a positive standardized design effect and power in (.5,1).');const z=zCritical(alpha),zp=normalInv(target),perGroup=Math.ceil(2*(z+zp)**2/d**2),curve=Array.from({length:50},(_,i)=>{const n=4+i*4,mu=d*Math.sqrt(n/2),power=normalCDF(-z-mu)+1-normalCDF(z-mu);return{n,power};});return{perGroup,d,alpha,target,curve,note:'Prospective normal approximation for equal independent groups and known common SD; not observed post-hoc power or exact t-test planning.'};}
/** CSV/TSV reader with proper quoted-cell support and explicit missing cells. */
export function parseCSV(text,delimiter=','){assert(typeof text==='string'&&text.length<=5000000,'Input is limited to 5 MB.');assert([',',';','\t'].includes(delimiter),'Unsupported delimiter.');text=text.replace(/^\uFEFF/,'');const records=[],row=[];let cell='',quoted=false;for(let i=0;i<=text.length;i++){const ch=text[i]??'\n';if(ch==='"'){if(quoted&&text[i+1]==='"'){cell+='"';i++;}else if(!quoted&&cell.trim())throw Error('Unexpected quote in an unquoted field.');else quoted=!quoted;}else if(!quoted&&(ch===delimiter||ch==='\n'||ch==='\r')){row.push(cell.trim());cell='';if(ch!==delimiter){if(row.some(x=>x!==''))records.push(row.splice(0));else row.length=0;if(ch==='\r'&&text[i+1]==='\n')i++;}}else cell+=ch;}assert(!quoted,'Unclosed CSV quotation.');assert(records.length>=2,'A header and at least one data row are required.');const columns=records.shift();assert(columns.length<=40&&columns.every(x=>x.length>0)&&new Set(columns).size===columns.length,'Use unique non-empty column names; maximum 40 columns.');assert(records.length<=50000,'At most 50,000 rows are accepted.');const rows=records.map((r,i)=>{assert(r.length===columns.length,`Row ${i+2}: expected ${columns.length} fields, found ${r.length}.`);return r.map(v=>v===''||/^(na|nan|null)$/i.test(v)?null:v);});return{columns,rows};}
export function numeric(v){if(v===null||v===undefined||v==='')return null;const n=Number(v);return Number.isFinite(n)?n:null;}
export function selectRows(table,cols,{missing='exclude'}={}){assert(cols.every(c=>Number.isInteger(c)&&c>=0&&c<table.columns.length),'Select valid columns.');const kept=[],excluded=[];for(let i=0;i<table.rows.length;i++){const values=cols.map(c=>numeric(table.rows[i][c]));if(values.some(v=>v===null)){excluded.push(i+2);if(missing==='error')throw Error(`Missing or non-numeric value in selected columns at CSV row ${i+2}.`);}else kept.push({row:i+2,values,raw:table.rows[i]});}return{kept,excluded,n:kept.length,total:table.rows.length};}
export function matchPairs(rows,groupIndex,idIndex,a,b,yIndex){assert(idIndex>=0,'Paired analysis requires an explicit subject/pair ID.');const ids=new Map();for(const row of rows){const g=row[groupIndex];if(g!==a&&g!==b)continue;const id=row[idIndex];assert(id!==null&&id!==undefined&&id!=='','Every paired row needs an ID.');if(!ids.has(id))ids.set(id,{});const pair=ids.get(id);assert(!(g in pair),`Duplicate observation for pair ${id}, group ${g}. Aggregate replicates explicitly first.`);pair[g]=numeric(row[yIndex]);}const A=[],B=[],kept=[],excluded=[];for(const[id,pair]of ids){if(pair[a]===null||pair[b]===null||pair[a]===undefined||pair[b]===undefined){excluded.push(id);continue;}A.push(pair[a]);B.push(pair[b]);kept.push(id);}return{a:A,b:B,ids:kept,excluded};}
export function summarizeReplicates(rows,idIndex,timeIndex,yIndex,metric='final'){assert(idIndex>=0&&timeIndex>=0,'Select run/subject ID and time.');const ids=new Map();for(const row of rows){const id=row[idIndex],t=numeric(row[timeIndex]),y=numeric(row[yIndex]);assert(id!==null&&id!==''&&t!==null&&y!==null,'Replicate summary refuses missing ID/time/value: resolve missingness explicitly.');if(!ids.has(id))ids.set(id,[]);ids.get(id).push([t,y]);}const summary=[...ids].map(([id,a])=>{a.sort((a,b)=>a[0]-b[0]);assert(a.every((x,i)=>!i||x[0]>a[i-1][0]),`Repeated times in run ${id}; one value per run and time is required.`);const x=a.map(p=>p[0]),y=a.map(p=>p[1]);const value=metric==='mean'?mean(y):metric==='max'?Math.max(...y):metric==='auc'?(clean(x,2),sum(y.slice(1).map((v,i)=>(v+y[i])*(x[i+1]-x[i])/2))):y.at(-1);return{id,start:x[0],end:x.at(-1),n:a.length,value};});assert(summary.every(r=>r.end===summary[0].end&&(metric!=='auc'||r.start===summary[0].start)),'Run time windows differ: use a common endpoint/window before comparing summaries.');return{metric,runs:summary,values:summary.map(x=>x.value),note:'One summary per run. Inference requires independent run seeds or independent experimental units, not repeated states.'};}
