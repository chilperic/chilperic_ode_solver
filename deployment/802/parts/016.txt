export function correlation(x,y,method='pearson',alpha=.05){clean(x,4);clean(y,4);assert(x.length===y.length,'Aligned observations required.');const a=method==='spearman'?rank(x):x,b=method==='spearman'?rank(y):y,mx=mean(a),my=mean(b),xx=sum(a.map(v=>(v-mx)**2)),yy=sum(b.map(v=>(v-my)**2));assert(xx>0&&yy>0,'Correlation is undefined for a constant variable.');const r=Math.max(-1,Math.min(1,sum(a.map((v,i)=>(v-mx)*(b[i]-my)))/Math.sqrt(xx*yy))),df=x.length-2,t=Math.abs(r)>=1?Math.sign(r)*Infinity:r*Math.sqrt(df/(1-r*r)),p=Math.abs(r)>=1?0:tP(t,df),z=zCritical(alpha),ci=method==='pearson'?(Math.abs(r)>=1?[r,r]:[Math.tanh(Math.atanh(r)-z/Math.sqrt(x.length-3)),Math.tanh(Math.atanh(r)+z/Math.sqrt(x.length-3))]):null;return{r,n:x.length,df,t:Number.isFinite(t)?t:null,p,ci,method,note:method==='pearson'?'Fisher-z approximate confidence interval; independent bivariate observations.':'Average ranks with ties; asymptotic t approximation for Spearman p-value. No Pearson CI is reused.'};}
export function permutation(a,b,{paired:isPaired=false,reps=2000,seed=42}={}){clean(a,2);clean(b,2);assert(Number.isInteger(reps)&&reps>=99&&reps<=20000,'Use 99–20,000 permutations.');assert((a.length+b.length)*reps<=8e6,'Permutation work budget exceeded; reduce samples or permutations.');const r=rng(seed),observed=mean(a)-mean(b),samples=[];let extreme=0;const all=a.concat(b);if(isPaired)assert(a.length===b.length,'Paired permutations require aligned pairs.');for(let k=0;k<reps;k++){let v;if(isPaired)v=mean(a.map((x,i)=>(x-b[i])*(r()<.5?-1:1)));else{const s=all.slice();for(let j=s.length-1;j>0;j--){const i=Math.floor(r()*(j+1));[s[j],s[i]]=[s[i],s[j]];}v=mean(s.slice(0,a.length))-mean(s.slice(a.length));}samples.push(v);if(Math.abs(v)>=Math.abs(observed)-1e-12)extreme++;}const p=(extreme+1)/(reps+1);return{estimate:observed,p,mcSE:Math.sqrt(p*(1-p)/(reps+1)),extreme,reps,seed,samples,scheme:isPaired?'Within-pair sign flips; symmetric differences / randomized pairing null.':'Exchangeable labels; tests a distributional/randomization null, not solely equal means with unequal variances.'};}
export function bootstrap(a,{reps=2000,seed=42,statistic='mean',alpha=.05,block=1}={}){clean(a,3);assert(Number.isInteger(reps)&&reps>=99&&reps<=20000,'Use 99–20,000 bootstrap draws.');assert(Number.isInteger(block)&&block>=1&&block<=Math.floor(a.length/2),'Block length must be between 1 and n/2.');assert(a.length*reps<=8e6,'Bootstrap work budget exceeded.');const fn=statistic==='median'?median:mean,r=rng(seed),samples=[];for(let b=0;b<reps;b++){const v=[];while(v.length<a.length){const i=Math.floor(r()*(a.length-block+1));v.push(...a.slice(i,i+block));}samples.push(fn(v.slice(0,a.length)));}return{estimate:fn(a),se:sd(samples),ci:[quantile(samples,alpha/2),quantile(samples,1-alpha/2)],bias:mean(samples)-fn(a),samples,reps,seed,block,statistic,alpha,scheme:block===1?'IID percentile bootstrap of observations.':'Overlapping moving-block bootstrap of ordered observations; stationarity and chosen block length are assumptions.'};}
function transpose(A){return A[0].map((_,j)=>A.map(r=>r[j]));}
function mm(A,B){return A.map(row=>B[0].map((_,j)=>sum(row.map((v,k)=>v*B[k][j]))));}
/** Reorthogonalized modified Gram-Schmidt QR, limited to 8 predictors. */
export function regress(X,y,{alpha=.05,covariance='classical'}={}){clean(y,4);assert(X.length===y.length&&X[0]?.length>=1&&X[0].length<=8,'Provide 1–8 aligned predictors.');const n=y.length,p=X[0].length+1;assert(n>p+1,'More independent observations than coefficients +1 are required.');assert(X.every(row=>row.length===p-1&&row.every(Number.isFinite)),'Non-finite predictor.');const A=X.map(r=>[1,...r]),cols=transpose(A),Q=[],R=Array.from({length:p},()=>Array(p).fill(0));for(let j=0;j<p;j++){let v=cols[j].slice();const scale=Math.sqrt(sum(v.map(x=>x*x)));for(let pass=0;pass<2;pass++)for(let k=0;k<j;k++){const c=sum(Q[k].map((x,i)=>x*v[i]));R[k][j]+=c;v=v.map((x,i)=>x-c*Q[k][i]);}const norm=Math.sqrt(sum(v.map(x=>x*x)));assert(norm>1e-10*Math.max(1,scale),'Predictors are rank deficient or severely collinear; remove redundant predictors or rescale.');R[j][j]=norm;Q.push(v.map(x=>x/norm));}const qty=Q.map(q=>sum(q.map((x,i)=>x*y[i]))),coef=Array(p);for(let i=p-1;i>=0;i--)coef[i]=(qty[i]-sum(R[i].slice(i+1).map((v,k)=>v*coef[i+k+1])))/R[i][i];const pred=A.map(row=>sum(row.map((x,i)=>x*coef[i]))),resid=y.map((v,i)=>v-pred[i]),sse=sum(resid.map(x=>x*x)),m=mean(y),sst=sum(y.map(x=>(x-m)**2));assert(sst>0,'Constant response: regression inference is undefined.');const df=n-p,mse=sse/df,invR=Array.from({length:p},()=>Array(p).fill(0));for(let j=0;j<p;j++)for(let i=p-1;i>=0;i--)invR[i][j]=((i===j?1:0)-sum(R[i].slice(i+1).map((x,k)=>x*invR[i+k+1][j])))/R[i][i];const inv=mm(invR,transpose(invR)),leverage=Array.from({length:n},(_,i)=>sum(Q.map(q=>q[i]**2))),estimable=mse>1e-24*sst/n;let cov=inv.map(row=>row.map(x=>x*mse));if(covariance==='hc3'){assert(leverage.every(x=>x<1-1e-10),'HC3 covariance undefined at unit leverage.');const meat=Array.from({length:p},(_,j)=>Array.from({length:p},(_,k)=>sum(A.map((a,i)=>a[j]*a[k]*resid[i]**2/(1-leverage[i])**2))));cov=mm(mm(inv,meat),inv);}const q=tCritical(alpha,df),se=cov.map((row,i)=>Math.sqrt(Math.max(0,row[i]))),coefficients=coef.map((estimate,i)=>({term:i===0?'Intercept':'x'+i,estimate,se:estimable?se[i]:null,t:estimable&&se[i]>0?estimate/se[i]:null,p:estimable&&se[i]>0?tP(estimate/se[i],df):null,ci:estimable?[estimate-q*se[i],estimate+q*se[i]]:null})),confidence=A.map((a,i)=>{const variance=sum(a.map((x,j)=>x*sum(a.map((v,k)=>v*cov[j][k]))));const d=q*Math.sqrt(Math.max(0,variance));return [pred[i]-d,pred[i]+d];}),prediction=covariance==='classical'?confidence.map((ci,i)=>{const d=Math.sqrt((ci[1]-pred[i])**2+q*q*mse);return[pred[i]-d,pred[i]+d];}):null;return{n,p,df,coefficients,pred,resid,leverage,cook:resid.map((r,i)=>mse>0?r*r/(p*mse)*leverage[i]/(1-leverage[i])**2:null),standardized:resid.map((r,i)=>mse>0?r/Math.sqrt(mse*Math.max(1e-12,1-leverage[i])):null),r2:1-sse/sst,adjR2:1-(sse/sst)*(n-1)/df,rmse:Math.sqrt(sse/n),residualSD:Math.sqrt(mse),confidence,prediction,covariance,alpha,estimable,durbinWatson:sum(resid.slice(1).map((x,i)=>(x-resid[i])**2))/(sse||1),note:covariance==='hc3'?'HC3 heteroskedasticity-robust SE with approximate t inference; not cluster-robust. Prediction intervals withheld.':'Classical OLS inference assumes independent errors, constant variance and a correctly specified conditional mean. Near-perfect fits receive no inferential p-values.'};}
export function wilson(successes,n,alpha=.05){assert(Number.isInteger(n)&&n>0&&Number.isInteger(successes)&&successes>=0&&successes<=n,'Binomial counts must be integers with 0≤successes≤n.');const p=successes/n,z=zCritical(alpha),den=1+z*z/n,c=(p+z*z/(2*n))/den,h=z*Math.sqrt(p*(1-p)/n+z*z/(4*n*n))/den;return{estimate:p,n,successes,ci:[Math.max(0,c-h),Math.min(1,c+h)]};}
function logChoose(n,k){return k<0||k>n?-Infinity:logGamma(n+1)-logGamma(k+1)-logGamma(n-k+1);}
export function contingency(table,alpha=.05){assert(table.length>=2&&table[0].length>=2,'At least a 2×2 table is required.');assert(table.every(r=>r.length===table[0].length&&r.every(x=>Number.isInteger(x)&&x>=0)),'Frequency cells must be nonnegative integers.');const row=table.map(sum),col=transpose(table).map(sum),N=sum(row);assert(row.every(x=>x>0)&&col.every(x=>x>0),'Remove empty margins before testing.');const expected=row.map(r=>col.map(c=>r*c/N)),chi2=sum(table.map((r,i)=>sum(r.map((x,j)=>(x-expected[i][j])**2/expected[i][j])))),df=(row.length-1)*(col.length-1);const result={N,expected,chi2,df,p:chiP(chi2,df),cramersV:Math.sqrt(chi2/(N*Math.min(row.length-1,col.length-1))),smallExpected:expected.flat().filter(x=>x<5).length,note:'Pearson chi-square without continuity correction; inspect expected counts. Row 1 is group A, column 1 denotes the event.'};if(row.length===2&&col.length===2){const [[a,b],[c,d]]=table;assert(N<=200000,'Fisher enumeration budget exceeded.');const logp=x=>logChoose(col[0],x)+logChoose(col[1],row[0]-x)-logChoose(N,row[0]),lp=logp(a),lo=Math.max(0,row[0]-col[1]),hi=Math.min(row[0],col[0]);let fisher=0;for(let x=lo;x<=hi;x++)if(logp(x)<=lp+1e-10)fisher+=Math.exp(logp(x));const A=wilson(a,a+b,alpha),B=wilson(c,c+d,alpha),diff=A.estimate-B.estimate,corrected=[a,b,c,d].some(x=>x===0),cells=[a,b,c,d].map(x=>x+(corrected?.5:0)),logOR=Math.log(cells[0]*cells[3]/(cells[1]*cells[2])),se=Math.sqrt(sum(cells.map(x=>1/x))),z=zCritical(alpha);result.fisherP=Math.min(1,fisher);result.oddsRatio=b*c>0?a*d/(b*c):null;result.logOddsRatioCI=[Math.exp(logOR-z*se),Math.exp(logOR+z*se)];result.oddsRatioCICorrection=corrected?'Haldane–Anscombe +0.5 to all four cells for the interval; raw OR is retained separately.':'No continuity correction.';result.riskDifference=diff;result.riskDifferenceCI=[diff-Math.sqrt((A.estimate-A.ci[0])**2+(B.ci[1]-B.estimate)**2),diff+Math.sqrt((A.ci[1]-A.estimate)**2+(B.estimate-B.ci[0])**2)];result.risks=[A,B];}return result;}
