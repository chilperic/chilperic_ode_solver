 if(id==='scatter'||id==='phase'){traces=[dots(x,y,'Observations',{marker:{size:6,opacity:.8}})];if(id==='phase')traces=[line(x,y,'Selected states',{mode:['trajectory','response'].includes(a.options.design)?'lines+markers':'markers',connectgaps:false,marker:{size:4}})];if(id==='scatter'&&r.pred&&r.p===2){const idx=fullX.map((_,i)=>i).sort((i,j)=>fullX[i]-fullX[j]);if(r.confidence){traces.push(line(idx.map(i=>fullX[i]),idx.map(i=>r.confidence[i][1]),'Mean upper',{showlegend:false,line:{width:0}}),line(idx.map(i=>fullX[i]),idx.map(i=>r.confidence[i][0]),'Mean confidence band',{fill:'tonexty',fillcolor:settings.dark?'rgba(86,180,233,.2)':'rgba(0,114,178,.14)',line:{width:0}}));}traces.push(line(idx.map(i=>fullX[i]),idx.map(i=>r.pred[i]),'Fitted conditional mean'));if(settings.prediction&&r.prediction)for(const k of [0,1])traces.push(line(idx.map(i=>fullX[i]),idx.map(i=>r.prediction[i][k]),k?'Prediction upper':'Prediction lower',{line:{dash:'dot'},showlegend:k===0}));if(animation==='residuals'){const count=Math.ceil(frame*y.length);for(let i=0;i<count;i++)traces.push(line([fullX[i],fullX[i]],[fullY[i],r.pred[i]],'Residual',{line:{color:colors[1],width:1.4},showlegend:false}));caption='Orange segments are observed minus fitted; the fitted model is unchanged during animation.';}}if(clip){layout.xaxis.range=range(fullX);layout.yaxis.range=range(fullY);caption=`${n} of ${N} recorded rows displayed. All numerical summaries still refer to the full run.`;}}
 if(id==='phase3d'){const zIndex=Number(settings.zColumn??a.audit.numericColumns.find(i=>i!==a.options.x&&i!==a.options.y));S.assert(Number.isInteger(zIndex)&&zIndex>=0&&zIndex<a.table.columns.length&&zIndex!==a.options.x&&zIndex!==a.options.y,'Select a distinct third numeric coordinate.');const raw=a.table.rows.map(r=>[a.options.x,a.options.y,zIndex].map(i=>S.numeric(r[i]))),v=clip?raw.slice(0,Math.max(1,Math.ceil(raw.length*frame))):raw;traces=[{type:'scatter3d',x:v.map(x=>x[0]),y:v.map(x=>x[1]),z:v.map(x=>x[2]),connectgaps:false,mode:['trajectory','response'].includes(a.options.design)?'lines+markers':'markers',marker:{size:2},name:'Selected states'}];layout.scene={xaxis:{title:a.table.columns[a.options.x]},yaxis:{title:a.table.columns[a.options.y]},zaxis:{title:a.table.columns[zIndex]}};caption='3D projection of selected columns. Spatial-looking coordinates do not imply anatomical space.';}
 if(id==='trajectory'){const xi=a.options.x,yi=a.options.y,rows=a.table.rows;let data=clip?rows.slice(0,Math.max(1,Math.ceil(rows.length*frame))):rows;traces=[line(data.map(r=>S.numeric(r[xi])),data.map(r=>S.numeric(r[yi])),'Stored response',{connectgaps:false,mode:'lines+markers',marker:{size:4}})];xlabel=a.table.columns[xi];ylabel=a.table.columns[yi];layout.xaxis.range=range(rows.map(r=>S.numeric(r[xi])));layout.yaxis.range=range(rows.map(r=>S.numeric(r[yi])));caption=(a.options.design==='response'?'Independent parameter/equilibrium samples, not physical time. ':'Recorded sequence, not a newly computed simulation. ')+'Missing or infeasible states remain visible gaps.';}
 if(id==='residual'){traces=[dots(r.pred,r.resid,'Residuals')];xlabel='Fitted response';ylabel='Observed − fitted';shapes.push({type:'line',xref:'paper',x0:0,x1:1,y0:0,y1:0,line:{dash:'dash',color:colors[1]}});caption=r.note;}
 if(id==='residual-qq'){const q=qq(r.resid);traces=[dots(q.x,q.y,'Residual quantiles'),line([q.x[0],q.x.at(-1)],[q.m+q.sd*q.x[0],q.m+q.sd*q.x.at(-1)],'Normal reference',{line:{dash:'dash'}})];xlabel='Standard normal quantile';ylabel='Residual';}
 if(id==='scale'){traces=[dots(r.pred,r.standardized.map(x=>x===null?null:Math.sqrt(Math.abs(x))),'Residual scale')];xlabel='Fitted response';ylabel='√|standardized residual|';caption='A pattern suggests nonconstant residual spread; this is not an automatic model-selection rule.';}
 if(id==='influence'){traces=[dots(r.leverage,r.cook,'Observations',{text:r.leverage.map((_,i)=>'Analysis row '+(i+1)),marker:{size:7}})];xlabel='Leverage';ylabel='Cook distance';caption='Investigate influential observations; do not delete them automatically.';}
 if(id==='acf'){traces=[{x:r.acf.map(x=>x.lag),y:r.acf.map(x=>x.r),type:'bar',name:'ACF'}];xlabel='Lag in observations';ylabel='Autocorrelation';caption='No white-noise confidence band is imposed on an arbitrary mechanistic trajectory.';}
 if(id==='replicate'){traces=[dots(r.runs.map(x=>x.id),r.values,'Run summary')];xlabel='Independent run ID';ylabel=r.metric+' '+a.units.y;caption=r.note;}
 if(id==='resampling'){const b=r.samples?r:r.bootstrap,values=(animation==='resampling'?b.samples.slice(0,Math.max(1,Math.ceil(frame*b.samples.length))):b.samples);traces=[{x:values,type:'histogram',name:a.method==='permutation'?'Null contrasts':'Resampled estimates',nbinsx:35,marker:{color:colors[0],opacity:.75}}];const obs=b.estimate;shapes.push({type:'line',yref:'paper',y0:0,y1:1,x0:obs,x1:obs,line:{color:colors[1],width:3}});if(b.ci)for(const q of b.ci)shapes.push({type:'line',yref:'paper',y0:0,y1:1,x0:q,x1:q,line:{color:colors[2],dash:'dash'}});xlabel=a.method==='permutation'?'Contrast under null':'Resampled '+b.statistic;ylabel='Count';layout.xaxis.range=range(b.samples);caption=`${values.length}/${b.samples.length} stored draws shown; seed ${b.seed}. Orange = observed estimate. ${b.scheme}`;}
 if(id==='survival'){for(const[g,i]of r.groups.map((g,i)=>[g,i])){const time=[0,...g.rows.map(r=>r.time)],surv=[1,...g.rows.map(r=>r.survival)];traces.push(line(time,[1,...g.rows.map(r=>r.ci[1])],g.name+' upper',{line:{width:0,shape:'hv'},showlegend:false}),line(time,[1,...g.rows.map(r=>r.ci[0])],g.name+' interval',{fill:'tonexty',opacity:.18,line:{width:0,shape:'hv'},showlegend:false}),line(time,surv,g.name,{line:{shape:'hv',color:colors[i%colors.length]}}));const cens=g.rows.filter(x=>x.censored);traces.push(dots(cens.map(x=>x.time),cens.map(x=>x.survival),g.name+' censored',{showlegend:false,marker:{symbol:'line-ns-open',size:9,color:colors[i%colors.length]}}));}xlabel=a.units.y;ylabel='Event-free probability';layout.yaxis.range=[0,1.02];caption='Pointwise Greenwood log-log confidence bands; tick marks indicate right censoring.';}
 if(id==='risk'){traces=r.groups.map(g=>line(g.rows.map(r=>r.time),g.rows.map(r=>r.risk),g.name,{line:{shape:'hv'},mode:'lines+markers'}));xlabel=a.units.y;ylabel='At risk just before event/censor time';}
 if(id==='counts'||id==='expected'){traces=[{type:'heatmap',z:id==='counts'?p.counts:p.counts.map((row,i)=>row.map((v,j)=>v-r.expected[i][j])),x:p.outcomeLabels,y:p.groupLabels,colorscale:id==='counts'?'Blues':'RdBu',zmid:id==='expected'?0:undefined,text:p.counts,texttemplate:'%{z:.3g}',hoverongaps:false}];xlabel='Outcome category';ylabel='Group';caption=id==='counts'?'Observed integer counts, not normalized percentages.':'Signed observed minus expected counts; not standardized Pearson residuals.';}
 if(id==='power'){traces=[line(r.curve.map(x=>x.n),r.curve.map(x=>x.power),'Normal-approximation power')];xlabel='Independent observations per group';ylabel='Approximate power';layout.yaxis.range=[0,1];caption=r.note;}
 layout.xaxis.title.text=xlabel;layout.yaxis.title.text=ylabel;
 for(const axis of ['x','y'])if(settings[axis+'Scale']==='log'){const data=traces.flatMap(t=>t[axis]||[]).filter(v=>typeof v==='number');S.assert(data.every(v=>v===null||v>0),'Log scale requires strictly positive displayed values; no observations were silently removed.');layout[axis+'axis'].type='log';delete layout[axis+'axis'].range;}
 for(const t of traces){if(t.type==='scatter'){t.line={...t.line,width:t.line?.width===0?0:Number(settings.lineWidth||2)};t.marker={...t.marker,size:Array.isArray(t.marker?.size)?t.marker.size:Number(settings.pointSize||6)};}}
 return{data:traces,layout,caption,title};
}
export function defaultViews(a){if(a.method==='regression')return['scatter','residual'];if(a.method==='survival')return['survival','risk'];if(a.method==='categorical')return['counts','expected'];if(a.method==='planning')return['power','histogram'];if(a.method==='paired')return['paired','difference'];if(['permutation','bootstrap','block-bootstrap'].includes(a.method))return['resampling',a.plots.groups?'groups':'ecdf'];if(a.method==='series')return['trajectory',a.result.acf?.length?'acf':'histogram'];if(a.method==='replicates')return['replicate','histogram'];if(a.method==='endpoints')return['forest','histogram'];if(['welch','welch-anova','kruskal'].includes(a.method))return['groups',a.result.comparisons?.some(x=>x.ci)||a.result.ci?'forest':'group-box'];if(a.method==='correlation')return['scatter','qq'];if(a.options.design==='response'||a.options.design==='trajectory')return['trajectory','ecdf'];return['histogram','ecdf'];}
