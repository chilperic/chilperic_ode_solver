import * as S from './statistics.mjs';
import {VIEWS} from './methods.mjs';
export const PALETTES={accessible:['#0072B2','#D55E00','#009E73','#CC79A7','#8A6500','#56B4E9','#332288','#555555'],teal:['#176B78','#AC4B2A','#427B47','#735691','#926C13','#295591'],mono:['#19373F','#546D75','#7E959B','#394F59']};
const groups=a=>a.plots.groups||{'Values':a.plots.y};
export function availability(a,id){const r=a.result,p=a.plots;if(['histogram','density','ecdf','box','violin','qq'].includes(id))return p.y?.length?'': 'Needs numerical observations.';
 if(['groups','group-box','group-violin'].includes(id))return p.groups?'':'Select a group column.';
 if(id==='group-ci')return p.groups&&a.options.design==='independent'&&a.options.independence&&Object.values(p.groups).every(v=>v.length>=2&&S.sd(v)>0)?'':'Needs independent groups with positive variance; paired/time samples are not IID.';
 if(['paired','difference','agreement'].includes(id))return p.pairs?'':'Run a paired analysis with explicit IDs.';
 if(id==='forest')return r.ci||r.comparisons?.some(c=>c.ci)||r.effects||r.coefficients?'':'Needs estimated differences, coefficients or endpoint intervals.';
 if(['residual','residual-qq','scale','influence'].includes(id))return r.resid?'':'Run regression first.';
 if(['scatter','phase'].includes(id))return Array.isArray(p.x)&&Array.isArray(p.y)&&p.x.length>0&&p.x.length===p.y.length&&p.x.every(Number.isFinite)?'':'Select two aligned numeric variables.';
 if(id==='phase3d')return a.audit.numericColumns.length>=3&&a.audit.numericColumns.includes(a.options.x)&&a.audit.numericColumns.includes(a.options.y)?'':'Needs three numeric coordinates; choose x/y and third variable in figure settings.';
 if(id==='trajectory')return a.options.x>=0&&a.audit.numericColumns.includes(a.options.x)&&a.audit.numericColumns.includes(a.options.y)?'':'Select ordered numeric independent and response variables.';
 if(id==='acf')return r.acf?.length?'':'Needs a nonconstant, regularly sampled complete time series.';
 if(id==='replicate')return r.runs?'':'Summarize independent runs first.';
 if(id==='resampling')return r.samples||r.bootstrap?.samples?'':'Run bootstrap or permutation analysis first.';
 if(['survival','risk'].includes(id))return a.method==='survival'?'':'Run Kaplan–Meier analysis first.';
 if(['counts','expected'].includes(id))return r.expected?'':'Run categorical analysis first.';
 if(id==='power')return r.curve?'':'Run prospective design analysis first.';return 'Unavailable view.';
}
function qq(y){const n=y.length,m=S.mean(y),sd=n>1?S.sd(y):0;S.assert(sd>0,'Q–Q plot is undefined for constant values.');return{x:y.map((_,i)=>S.normalInv((i+.5)/n)),y:y.slice().sort((a,b)=>a-b),m,sd};}
const line=(x,y,name,extra={})=>({type:'scatter',x,y,mode:'lines',name,...extra});
const dots=(x,y,name,extra={})=>({type:'scatter',x,y,mode:'markers',name,...extra});
function range(a){const v=a.filter(Number.isFinite);if(!v.length)return[-1,1];const lo=Math.min(...v),hi=Math.max(...v),pad=(hi-lo)*.07||Math.abs(lo)*.1||1;return[lo-pad,hi+pad];}
export function plot(a,id,settings={},frame=1,animation='off'){
 const unavailable=availability(a,id);S.assert(!unavailable,unavailable);const p=a.plots,r=a.result,colors=PALETTES[settings.palette]||PALETTES.accessible,alpha=a.options.alpha,fullY=p.y||[],fullX=p.x||[],N=fullY.length,n=Math.max(1,Math.ceil(N*frame)),clip=animation==='reveal';let y=clip?fullY.slice(0,n):fullY,x=clip?fullX.slice(0,n):fullX,traces=[],caption='',title=VIEWS.find(v=>v.id===id)?.name||id,xlabel=a.units.x,ylabel=a.units.y,shapes=[];
 let layout={title:{text:settings.title||title,font:{size:Number(settings.fontSize||16)+2}},paper_bgcolor:settings.dark?'#172E35':'#FFFFFF',plot_bgcolor:settings.dark?'#172E35':'#FFFFFF',font:{family:'Arial, system-ui, sans-serif',size:Number(settings.fontSize||14),color:settings.dark?'#EAF2F3':'#19373F'},margin:{l:72,r:26,t:58,b:70},colorway:colors,showlegend:true,legend:{orientation:'h',y:-.22},xaxis:{title:{text:xlabel},automargin:true,gridcolor:settings.dark?'#35515A':'#E4EBE8',zeroline:false},yaxis:{title:{text:ylabel},automargin:true,gridcolor:settings.dark?'#35515A':'#E4EBE8',zeroline:false},hovermode:'closest',uirevision:id,shapes};
 if(['histogram','density'].includes(id)){traces=[{x:y,type:'histogram',name:a.units.y,histnorm:settings.density||id==='density'?'probability density':'',nbinsx:Number(settings.bins)||undefined,marker:{color:colors[0],opacity:.65}}];xlabel=a.units.y;ylabel=id==='density'||settings.density?'Density':'Count';if(id==='density'){S.assert(fullY.length>=3&&S.sd(fullY)>0,'Density estimate needs at least three variable observations.');const bw=1.06*S.sd(fullY)*fullY.length**(-.2),lo=Math.min(...fullY)-3*bw,hi=Math.max(...fullY)+3*bw,g=Array.from({length:160},(_,i)=>lo+(hi-lo)*i/159);traces.push(line(g,g.map(t=>S.sum(y.map(v=>Math.exp(-.5*((t-v)/bw)**2)/(bw*Math.sqrt(2*Math.PI))))/y.length),'Gaussian density estimate'));caption=`Bandwidth ${bw.toPrecision(4)}; smoothing is a display estimate, not additional data.`;}}
 if(id==='ecdf'){const s=y.slice().sort((a,b)=>a-b);traces=[line(s,s.map((_,i)=>(i+1)/s.length),'Empirical CDF',{line:{shape:'hv'}})];xlabel=a.units.y;ylabel='Fraction ≤ x';layout.yaxis.range=[0,1];caption='A step for each observed value; ties retain their multiplicity.';}
 if(id==='box'||id==='violin'){S.assert(id!=='violin'||y.length>=5,'A violin density needs at least five observations.');traces=[{y,type:id,boxpoints:'all',points:'all',jitter:.18,name:a.units.y,box:{visible:true},meanline:{visible:true},marker:{size:5},pointpos:0}];xlabel='';}
 if(id==='qq'){const q=qq(y);traces=[dots(q.x,q.y,'Observed quantiles'),line([q.x[0],q.x.at(-1)],[q.m+q.sd*q.x[0],q.m+q.sd*q.x.at(-1)],'Fitted-normal reference',{line:{dash:'dash'}})];xlabel='Standard normal quantile';caption='A visual distribution diagnostic, not an automated normality pass/fail rule.';}
 if(['groups','group-box','group-violin','group-ci'].includes(id)){Object.entries(groups(a)).forEach(([name,v],i)=>{if(id==='groups')traces.push(dots(v.map(()=>name),v,name,{marker:{color:colors[i%colors.length],size:7,opacity:.8}}));else if(id==='group-ci'){const m=S.mean(v),q=S.tCritical(alpha,v.length-1)*S.sd(v)/Math.sqrt(v.length);traces.push(dots([name],[m],name,{error_y:{type:'data',array:[q],visible:true},marker:{size:10,color:colors[i%colors.length]}}));}else traces.push({y:v,name,type:id==='group-box'?'box':'violin',boxpoints:'all',points:'all',jitter:.2,meanline:{visible:true},box:{visible:true},marker:{color:colors[i%colors.length],size:5}});});xlabel='Group';caption=id==='group-ci'?`${(1-alpha)*100}% unadjusted pointwise mean t intervals; independent groups only.`:'Original observations are retained; no raw values are replaced by a mean-only bar.';}
 if(id==='paired'){const pair=p.pairs;traces=pair.a.map((v,i)=>line([r.groupA||'A',r.groupB||'B'],[v,pair.b[i]],String(pair.ids[i]),{mode:'lines+markers',showlegend:false,opacity:.55,line:{width:1.4,color:colors[0]},marker:{size:6}}));xlabel='Matched condition';caption=`${pair.ids.length} matched units. Lines connect the same ID, not neighboring rows.`;}
 if(id==='difference'){const diff=p.pairs.a.map((x,i)=>x-p.pairs.b[i]);traces=[dots(p.pairs.ids,diff,'Within-pair differences')];xlabel='Pair ID';ylabel=`${r.groupA||'A'} − ${r.groupB||'B'}`;shapes.push({type:'line',xref:'paper',x0:0,x1:1,y0:0,y1:0,line:{dash:'dash',color:colors[1]}});}
 if(id==='agreement'){const pair=p.pairs,d=pair.a.map((v,i)=>v-pair.b[i]),m=pair.a.map((v,i)=>(v+pair.b[i])/2),md=S.mean(d),s=S.sd(d);traces=[dots(m,d,'Pairs',{text:pair.ids})];for(const[v,name]of[[md,'Bias'],[md-1.96*s,'Lower agreement limit'],[md+1.96*s,'Upper agreement limit']])traces.push(line(range(m),[v,v],name,{line:{dash:'dash'}}));xlabel='Pair mean';ylabel='A − B';caption='Bland–Altman descriptive 95% limits = bias ±1.96 SD; not confidence intervals and not proof of agreement.';}
 if(id==='forest'){let rows=r.effects?r.effects.filter(x=>x.ci).map(x=>({...x,label:x.endpoint})):r.coefficients?r.coefficients.filter(x=>x.ci).map(x=>({...x,label:x.term})):r.comparisons?r.comparisons.filter(x=>x.ci).map(x=>({...x,label:x.a+' − '+x.b})):[{...r,label:r.groupA?r.groupA+' − '+r.groupB:a.methodName}];S.assert(rows.length,'No estimable intervals available.');traces=[dots(rows.map(x=>x.estimate),rows.map(x=>x.label),'Estimate',{error_x:{type:'data',symmetric:false,array:rows.map(x=>x.ci[1]-x.estimate),arrayminus:rows.map(x=>x.estimate-x.ci[0]),visible:true},marker:{size:9}})];xlabel='Estimate in outcome units';ylabel='';caption=`${(1-alpha)*100}% intervals. Unadjusted intervals are not simultaneous intervals; inspect adjusted p-values in the results table. Different endpoint units require separate interpretation.`;shapes.push({type:'line',yref:'paper',y0:0,y1:1,x0:0,x1:0,line:{dash:'dash',color:colors[1]}});}
