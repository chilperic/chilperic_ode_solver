/** Scientific Statistics — strict data and explicit design contracts. Pure ESM.
 * Deterministic computation; missing is never coerced to zero; no automatic tests.
 * Numerical references: SciPy statistics documentation; method notes in methods.mjs.
 */
export const VERSION='80.2.0';
export function assert(ok,message){if(!ok)throw new Error(message);}
export function clean(a,min=1){assert(Array.isArray(a)&&a.length>=min,`At least ${min} usable observations are required.`);assert(a.every(x=>typeof x==='number'&&Number.isFinite(x)),'Non-finite or non-numeric observations: select a missing-value policy first.');return a;}
export const sum=a=>a.reduce((s,x)=>s+x,0);
export const mean=a=>sum(clean(a))/a.length;
export function variance(a){clean(a,2);const m=mean(a);return sum(a.map(x=>(x-m)**2))/(a.length-1);}
export const sd=a=>Math.sqrt(variance(a));
export function quantile(a,p){clean(a);assert(p>=0&&p<=1,'Quantile probability must be in [0,1].');const s=a.slice().sort((a,b)=>a-b),t=(s.length-1)*p,i=Math.floor(t);return s[i]+(s[Math.min(i+1,s.length-1)]-s[i])*(t-i);}
export const median=a=>quantile(a,.5);
export function describe(a){clean(a);const m=mean(a),med=median(a);return{n:a.length,mean:m,median:med,sd:a.length>1?sd(a):null,sem:a.length>1?sd(a)/Math.sqrt(a.length):null,min:Math.min(...a),q1:quantile(a,.25),q3:quantile(a,.75),max:Math.max(...a),iqr:quantile(a,.75)-quantile(a,.25),mad:median(a.map(x=>Math.abs(x-med)))};}
export function rng(seed=42){assert(Number.isInteger(seed),'Seed must be an integer.');let a=seed>>>0;return()=>{a+=0x6D2B79F5;let t=a;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return((t^(t>>>14))>>>0)/4294967296;};}
export function normalRng(seed=42){const r=rng(seed);return()=>Math.sqrt(-2*Math.log(Math.max(r(),1e-15)))*Math.cos(2*Math.PI*r());}
export function logGamma(z){assert(z>0,'Gamma argument must be positive.');const c=[676.5203681218851,-1259.1392167224028,771.32342877765313,-176.61502916214059,12.507343278686905,-.13857109526572012,9.9843695780195716e-6,1.5056327351493116e-7];if(z<.5)return Math.log(Math.PI)-Math.log(Math.sin(Math.PI*z))-logGamma(1-z);z--;let x=.99999999999980993;c.forEach((a,i)=>x+=a/(z+i+1));const t=z+7.5;return .5*Math.log(2*Math.PI)+(z+.5)*Math.log(t)-t+Math.log(x);}
function betaCF(a,b,x){const tiny=1e-300;let c=1,d=1-(a+b)*x/(a+1);if(Math.abs(d)<tiny)d=tiny;d=1/d;let h=d;for(let m=1;m<=1000;m++){let aa=m*(b-m)*x/((a+2*m-1)*(a+2*m));d=1+aa*d;if(Math.abs(d)<tiny)d=tiny;c=1+aa/c;if(Math.abs(c)<tiny)c=tiny;d=1/d;h*=d*c;aa=-(a+m)*(a+b+m)*x/((a+2*m)*(a+2*m+1));d=1+aa*d;if(Math.abs(d)<tiny)d=tiny;c=1+aa/c;if(Math.abs(c)<tiny)c=tiny;d=1/d;const delta=d*c;h*=delta;if(Math.abs(delta-1)<3e-14)return h;}throw Error('Incomplete beta did not converge.');}
export function betaCDF(x,a,b){assert(a>0&&b>0,'Beta shape must be positive.');if(x<=0)return 0;if(x>=1)return 1;const k=Math.exp(logGamma(a+b)-logGamma(a)-logGamma(b)+a*Math.log(x)+b*Math.log1p(-x));return x<(a+1)/(a+b+2)?k*betaCF(a,b,x)/a:1-k*betaCF(b,a,1-x)/b;}
export function tP(t,df){assert(df>0,'Positive degrees of freedom required.');return betaCDF(df/(df+t*t),df/2,.5);}
export function tCritical(alpha,df){assert(alpha>0&&alpha<1&&df>0,'Invalid alpha or degrees of freedom.');let lo=0,hi=1;while(tP(hi,df)>alpha&&hi<1e14)hi*=2;for(let i=0;i<90;i++){const m=(lo+hi)/2;if(tP(m,df)>alpha)lo=m;else hi=m;}return (lo+hi)/2;}
export function fP(f,d1,d2){assert(d1>0&&d2>0&&f>=0,'Invalid F distribution inputs.');return betaCDF(d2/(d2+d1*f),d2/2,d1/2);}
export function gammaQ(a,x){assert(a>0&&x>=0,'Invalid gamma distribution inputs.');if(!x)return 1;const log=-x+a*Math.log(x)-logGamma(a);if(x<a+1){let t=1/a,s=t;for(let n=1;n<2000;n++){t*=x/(a+n);s+=t;if(Math.abs(t)<Math.abs(s)*1e-14)return Math.max(0,1-s*Math.exp(log));}}else{let b=x+1-a,c=1e300,d=1/b,h=d;for(let i=1;i<2000;i++){const an=-i*(i-a);b+=2;d=an*d+b;if(Math.abs(d)<1e-300)d=1e-300;c=b+an/c;if(Math.abs(c)<1e-300)c=1e-300;d=1/d;const del=d*c;h*=del;if(Math.abs(del-1)<1e-14)return Math.min(1,Math.max(0,Math.exp(log)*h));}}throw Error('Gamma tail did not converge.');}
export const chiP=(x,df)=>gammaQ(df/2,x/2);
export function normalCDF(z){const p=.5*gammaQ(.5,z*z/2);return z>=0?1-p:p;}
export function zCritical(alpha){assert(alpha>0&&alpha<1,'Invalid alpha.');let lo=0,hi=12;for(let i=0;i<90;i++){const m=(lo+hi)/2;if(2*(1-normalCDF(m))>alpha)lo=m;else hi=m;}return (lo+hi)/2;}
export function normalInv(p){assert(p>0&&p<1,'Normal quantile requires 0<p<1.');if(p===.5)return 0;return p>.5?zCritical(2*(1-p)):-zCritical(2*p);}
export function rank(a){clean(a);const idx=a.map((v,i)=>({v,i})).sort((a,b)=>a.v-b.v),r=Array(a.length);for(let i=0;i<idx.length;){let j=i+1;while(j<idx.length&&idx[j].v===idx[i].v)j++;for(let k=i;k<j;k++)r[idx[k].i]=(i+j+1)/2;i=j;}return r;}
export function adjust(ps,method='holm'){assert(['holm','bh','none'].includes(method),'Choose Holm, BH or unadjusted.');assert(ps.every(x=>Number.isFinite(x)&&x>=0&&x<=1),'Invalid p-values.');if(method==='none')return ps.slice();const idx=ps.map((p,i)=>({p,i})).sort((a,b)=>a.p-b.p),out=Array(ps.length),n=ps.length;let value=method==='holm'?0:1;const order=method==='holm'?idx.map((_,i)=>i):idx.map((_,i)=>n-i-1);for(const i of order){value=method==='holm'?Math.max(value,Math.min(1,idx[i].p*(n-i))):Math.min(value,idx[i].p*n/(i+1));out[idx[i].i]=Math.min(1,value);}return out;}
export function oneSample(a,mu0=0,alpha=.05){clean(a,2);assert(Number.isFinite(mu0),'Reference mean must be finite.');const d=describe(a),se=d.sd/Math.sqrt(d.n),df=d.n-1;assert(se>0,'Zero sample variance: a t statistic and confidence interval are not estimable.');const t=(d.mean-mu0)/se,c=tCritical(alpha,df);return{...d,estimate:d.mean-mu0,mu0,se,t,df,p:tP(t,df),ci:[d.mean-mu0-c*se,d.mean-mu0+c*se],meanCI:[d.mean-c*se,d.mean+c*se],alpha};}
export function welch(a,b,alpha=.05){clean(a,2);clean(b,2);const A=describe(a),B=describe(b),va=A.sd**2/A.n,vb=B.sd**2/B.n,se=Math.sqrt(va+vb);assert(se>0,'Both groups have zero variance: Welch inference is undefined.');const estimate=A.mean-B.mean,df=(va+vb)**2/(va**2/(A.n-1)+vb**2/(B.n-1)),t=estimate/se,q=tCritical(alpha,df),dof=A.n+B.n-2,pooled=Math.sqrt(((A.n-1)*A.sd**2+(B.n-1)*B.sd**2)/dof),J=Math.exp(logGamma(dof/2)-.5*Math.log(dof/2)-logGamma((dof-1)/2));return{A,B,estimate,se,t,df,p:tP(t,df),ci:[estimate-q*se,estimate+q*se],cohenD:estimate/pooled,hedgesG:J*estimate/pooled,alpha,direction:'Group A minus group B; g uses pooled SD only for effect standardization.'};}
export function paired(a,b,alpha=.05){clean(a,2);clean(b,2);assert(a.length===b.length,'Pairs must have equal lengths; match them by ID, not row order.');const differences=a.map((x,i)=>x-b[i]),r=oneSample(differences,0,alpha);return{...r,A:describe(a),B:describe(b),differences,dz:r.estimate/r.sd,direction:'Within-pair A minus B; n is the number of pairs.'};}
export function pairwise(groups,alpha=.05,correction='holm'){const names=Object.keys(groups);const rows=[];for(let i=0;i<names.length;i++)for(let j=i+1;j<names.length;j++)rows.push({a:names[i],b:names[j],...welch(groups[names[i]],groups[names[j]],alpha)});const adj=adjust(rows.map(x=>x.p),correction);return rows.map((x,i)=>({...x,pAdjusted:adj[i],correction,ciScope:'Unadjusted pairwise CI; adjusted p-values use the declared comparison family.'}));}
export function welchAnova(groups){const names=Object.keys(groups),k=names.length;assert(k>=2,'At least two groups are required.');const d=names.map(n=>describe(clean(groups[n],2)));assert(d.every(x=>x.sd>0),'Welch ANOVA requires positive within-group variance in every group.');const w=d.map(x=>x.n/x.sd**2),W=sum(w),m=sum(d.map((x,i)=>w[i]*x.mean))/W,B=sum(d.map((x,i)=>(1-w[i]/W)**2/(x.n-1))),df1=k-1,df2=(k*k-1)/(3*B),F=(sum(d.map((x,i)=>w[i]*(x.mean-m)**2))/df1)/(1+2*(k-2)*B/(k*k-1));return{F,df1,df2,p:fP(F,df1,df2),groups:names.map((n,i)=>({name:n,...d[i]})),weightedMean:m};}
export function anova(groups){const entries=Object.entries(groups),k=entries.length;assert(k>=2,'At least two groups required.');entries.forEach(([_,a])=>clean(a,2));const all=entries.flatMap(([_,a])=>a),m=mean(all),ssb=sum(entries.map(([_,a])=>a.length*(mean(a)-m)**2)),ssw=sum(entries.map(([_,a])=>(a.length-1)*variance(a))),df1=k-1,df2=all.length-k;assert(ssw>0,'No within-group variance; ordinary ANOVA is degenerate.');const F=(ssb/df1)/(ssw/df2);return{F,df1,df2,p:fP(F,df1,df2),etaSquared:ssb/(ssb+ssw),omegaSquared:(ssb-df1*ssw/df2)/(ssb+ssw+ssw/df2),ssBetween:ssb,ssWithin:ssw};}
export function brownForsythe(groups){const deviations=Object.fromEntries(Object.entries(groups).map(([n,a])=>{const m=median(a);return[n,a.map(x=>Math.abs(x-m))];}));return anova(deviations);}
export function kruskal(groups,correction='holm'){const names=Object.keys(groups),ns=names.map(n=>clean(groups[n],2).length),all=names.flatMap(n=>groups[n]),N=all.length;assert(names.length>=2,'At least two groups required.');const ranks=rank(all),sums=[],counts=new Map();all.forEach(x=>counts.set(x,(counts.get(x)||0)+1));let offset=0;for(const n of ns){sums.push(sum(ranks.slice(offset,offset+n)));offset+=n;}const ties=sum([...counts.values()].map(n=>n**3-n)),C=1-ties/(N**3-N);assert(C>0,'All observations tied: rank inference is undefined.');const H=Math.max(0,(12/(N*(N+1))*sum(sums.map((r,i)=>r*r/ns[i]))-3*(N+1))/C),rows=[];for(let i=0;i<ns.length;i++)for(let j=i+1;j<ns.length;j++){const z=(sums[i]/ns[i]-sums[j]/ns[j])/Math.sqrt((N*(N+1)/12-ties/(12*(N-1)))*(1/ns[i]+1/ns[j]));rows.push({a:names[i],b:names[j],z,p:gammaQ(.5,z*z/2)});}const padj=adjust(rows.map(r=>r.p),correction);return{H,df:names.length-1,p:chiP(H,names.length-1),tieCorrection:C,comparisons:rows.map((r,i)=>({...r,pAdjusted:padj[i]})),epsilonSquared:Math.max(0,(H-names.length+1)/(N-names.length)),note:'Asymptotic Kruskal–Wallis with tie correction; Dunn post-hoc rank contrasts. Distributional, not automatically median, differences.'};}
